# 一、文件预览

```ts
/**
 * 把图片文件读成 Base64 预览地址。
 */
function readImageAsBase64(file: File) {
  return new Promise<string>((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(String(reader.result))
    reader.onerror = () => reject(reader.error)
    // 把二进制文件转成可直接给 img 使用的 DataURL。
    reader.readAsDataURL(file)
  })
}
```

- 原理总结：`FileReader` 把本地文件读成浏览器可展示的字符串结果。
- 注意点：大文件不适合长期保留 Base64，体积会变大。

# 二、对象 URL 与下载

```ts
/**
 * 为文件创建临时预览地址。
 */
function createObjectURL(file: File) {
  return URL.createObjectURL(file)
}

/**
 * 下载 Blob 数据。
 */
function downloadBlob(filename: string, blob: Blob) {
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = filename
  a.click()
  // 下载后释放临时地址，避免内存泄漏。
  URL.revokeObjectURL(url)
}
```

- 原理总结：对象 URL 是浏览器为内存中的文件数据生成的临时访问地址。
- 注意点：用完一定要释放 URL。

# 三、拖拽上传

```ts
/**
 * 绑定拖拽上传区域。
 */
function bindDropZone(element: HTMLElement, onFiles: (files: File[]) => void) {
  element.addEventListener('dragover', (event) => {
    // 阻止浏览器默认打开文件行为。
    event.preventDefault()
  })

  element.addEventListener('drop', (event) => {
    event.preventDefault()
    const files = Array.from(event.dataTransfer?.files ?? [])
    onFiles(files)
  })
}
```

- 原理总结：通过阻止默认行为，把拖拽文件动作变成可控上传入口。
- 注意点：拖拽上传前仍然要做大小和类型校验。

# 四、文件切片

```ts
/**
 * 按固定大小切分文件。
 */
function sliceFile(file: File, chunkSize = 1024 * 1024) {
  const chunks: Blob[] = []
  let start = 0

  while (start < file.size) {
    chunks.push(file.slice(start, start + chunkSize))
    start += chunkSize
  }

  return chunks
}
```

- 原理总结：把大文件切成小块，便于分片上传和断点续传。
- 注意点：切片只是前端第一步，真正续传还要后端配合记录进度。
